export const ArcadeVideoButton = ({ id, title = "Watch interactive demo", label = "Watch Video" }) => {
  const iframeSrc = `https://demo.arcade.software/${id}?embed&embed_custom&show_copy_link=true`;

  const handleClick = () => {
    if (typeof document === 'undefined') return;
    const arcadeIframe = document.querySelector(`iframe[data-arcade-id="${id}"]`);
    if (arcadeIframe && arcadeIframe.contentWindow) {
      arcadeIframe.contentWindow.postMessage({ event: 'request-popout-open' }, '*');
    }
  };

  return (
    <div>
      <button
        onClick={handleClick}
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          gap: '8px',
          backgroundColor: '#3e47ec',
          color: '#ffffff',
          border: 'none',
          padding: '10px 18px',
          borderRadius: '8px',
          fontSize: '15px',
          fontWeight: '600',
          cursor: 'pointer',
        }}
      >
        <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="white">
          <polygon points="5,3 19,12 5,21" />
        </svg>
        {label}
      </button>

      <iframe
        data-arcade-id={id}
        src={iframeSrc}
        title={title}
        frameBorder="0"
        loading="lazy"
        allowFullScreen
        allow="clipboard-write"
        style={{ position: 'fixed', top: 0, left: 0, width: '100%', height: 0, colorScheme: 'light' }}
      />
    </div>
  );
};
